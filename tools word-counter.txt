    <div class="tool-box">
      <h1>Word Counter</h1>
      <textarea id="text" rows="8" placeholder="Type or paste your text here..."></textarea>
      <div class="result" id="result">
        Words: 0 | Characters: 0 | Sentences: 0 | Reading time: 0 min
      </div>
    </div>
  </div>
  <script>
    const text = document.getElementById('text');
    const result = document.getElementById('result');
    text.addEventListener('input', () => {
      const val = text.value;
      const words = val.trim() === '' ? 0 : val.trim().split(/\s+/).length;
      const chars = val.length;
      const sentences = (val.match(/[.!?]+/g) || []).length;
      const reading = Math.ceil(words / 200);
      result.innerHTML = `Words: <b>${words}</b> | Characters: <b>${chars}</b> | Sentences: <b>${sentences}</b> | Reading time: <b>${reading}</b> min`;
    });
  </script>
